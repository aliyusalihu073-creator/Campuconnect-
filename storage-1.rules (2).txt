rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {

    function isAdmin() {
      return request.auth != null &&
        request.auth.token.email in [
          'aliyusalihu@gmail.com',
          'admin@campusconnect.com',
          'aliyusalihu073@gmail.com'
        ];
    }

    function isSignedIn() {
      return request.auth != null;
    }

    // News photos/videos uploaded from admin-news.html — admin only
    match /news-images/{fileName} {
      allow read: if true;
      allow write: if isAdmin() && request.resource.size < 20 * 1024 * 1024;
    }
    match /news-videos/{fileName} {
      allow read: if true;
      allow write: if isAdmin() && request.resource.size < 20 * 1024 * 1024;
    }

    // Media hub uploads from admin-media.html — admin only
    match /media/{fileName} {
      allow read: if true;
      allow write: if isAdmin() && request.resource.size < 20 * 1024 * 1024;
    }

    // Handouts from admin-handout.html — admin only, students can view
    match /handouts/{fileName} {
      allow read: if isSignedIn();
      allow write: if isAdmin() && request.resource.size < 20 * 1024 * 1024;
    }

    // Documents — used by BOTH admin-docs-2.html (admin-only PDFs) and
    // chat.html (any student sharing a file in chat), so anyone signed in may
    // write here. If you want doc uploads restricted to admins only, tell me
    // and I'll split this into two separate folders instead.
    match /documents/{fileName} {
      allow read: if isSignedIn();
      allow write: if isSignedIn() && request.resource.size < 20 * 1024 * 1024;
    }

    // Chat photo/video attachments — any signed-in student
    match /chat/{fileName} {
      allow read: if isSignedIn();
      allow write: if isSignedIn() && request.resource.size < 20 * 1024 * 1024;
    }

    // Chat voice notes — any signed-in student
    match /voice/{fileName} {
      allow read: if isSignedIn();
      allow write: if isSignedIn() && request.resource.size < 20 * 1024 * 1024;
    }

    // Default deny everything else
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
